Persistence & Recovery

Write-Ahead Log (WAL)

Enable durable operation logging for crash recovery.

import "github.com/TollMesh/toll-mesh-store/persistence" // Create WAL with configuration config := persistence.WALConfig{ MaxSegmentSize: 100 * 1024 * 1024, // 100MB RotationTime: 24 * time.Hour, } wal, err := persistence.NewWriteAheadLog("/var/lib/toll/wal", config) if err != nil { log.Fatal(err) } // Log an operation entry := &persistence.WALEntry{ Timestamp: time.Now().UnixMilli(), Operation: "consume", Key: "user:123", Value: []byte("100"), } err = wal.Append(entry) if err != nil { log.Fatal(err) } // Start rotation wal.Start() defer wal.Stop()

Example 1: API Rate Limiting

Implement rate limiting for an API endpoint.

func apiHandler(w http.ResponseWriter, r *http.Request) { userID := r.Header.Get("X-User-ID") // Check rate limit: 100 requests per minute result, err := meshStore.Consume(r.Context(), fmt.Sprintf("api:%s", userID), 100, time.Minute) if err != nil { http.Error(w, "Internal error", http.StatusInternalServerError) return } if !result.OK { w.Header().Set("Retry-After", fmt.Sprintf("%d", result.ResetAt)) http.Error(w, "Rate limited", http.StatusTooManyRequests) return } // Process request w.WriteHeader(http.StatusOK) w.Write([]byte("Success")) }

Example 2: Replay Attack Prevention

Prevent replay attacks using nonce validation.

func validateRequest(r *http.Request) error { nonce := r.Header.Get("X-Nonce") if nonce == "" { return fmt.Errorf("missing nonce") } // Check if nonce was already used (5 minute window) seen, err := meshStore.Seen(r.Context(), fmt.Sprintf("nonce:%s", nonce), 5*time.Minute) if err != nil { return err } if seen { return fmt.Errorf("replay attack detected") } return nil }
import requests def validate_request(headers): nonce = headers.get('X-Nonce') if not nonce: raise ValueError("Missing nonce") # Check if nonce was already used response = requests.post('http://localhost:8080/seen', json={ 'key': f'nonce:{nonce}', 'ttl': 300000 # 5 minutes }) data = response.json() if data['seen']: raise ValueError("Replay attack detected") return True
const axios = require('axios'); async function validateRequest(headers) { const nonce = headers['x-nonce']; if (!nonce) { throw new Error("Missing nonce"); } // Check if nonce was already used const response = await axios.post('http://localhost:8080/seen', { key: `nonce:${nonce}`, ttl: 300000 // 5 minutes }); if (response.data.seen) { throw new Error("Replay attack detected"); } return true; }
